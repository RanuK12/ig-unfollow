Instagram Unfollow Tracker

Quickly identify and unfollow users who don't follow you back

// Instagram Unfollow Tracker
// Usage: Paste in browser console

(function() {
    "use strict";

    function getProfileUsername() {
        return document.querySelector('meta[property="og:title"]')?.content.split(' \-')[0];
    }

    function getFollowers() {
        return Array.from(document.querySelectorAll('a:not(.private)'))
            .filter(a => a.textContent.trim() === 'Following');
    }

    function unfollow(user) {
        user.click();
        setTimeout(() => {
            document.querySelector('button:has-text("Unfollow")')?.click();
        }, 1000);
    }

    function trackUnfollows() {
        const username = getProfileUsername();
        const follows = getFollowers();
        console.log(`\n✅ Tracking unfollows for @${username}...`);
        follows.forEach((user, index) => {
            setTimeout(() => {
                unfollow(user);
                console.log(`Unfollowed user #${index + 1}`);
            }, index * 2000);
        });
    }

    trackUnfollows();
})();

How to Use

  1. 1. Go to any Instagram profile
  2. 2. Open browser console (F12 or right-click > Inspect)
  3. 3. Paste the script above and press Enter
Warning: Instagram has rate limits. Use sparingly to avoid account restrictions.
